Filter for Contemporary Vagrants:

  # Returns: Bufflehead, Red-flanked Bluetail, Siberian Accentor, 
  #          Yellow-breasted Bunting, Baird's Sandpiper

Filter for Species Not Recorded in 20 Years:
  species = species[species['last_20yr_records'] == 0]
  # Returns: American Golden Plover, Upland Sandpiper, Yellow-billed Cuckoo, etc.

Filter for Species with Established Recent Pattern:
  species = species[species['last_10yr_records'] > 0]
